    \t
  • for each pixel, find which cell it's in; each cell is __cs(w) wide.
    • \t
  • find the hash at the bottom left of the cell mod 4.
    • \t
  • index into an array that decides the two points about which we're making circles
    • \t
  • draw the circles with some nice glow
    • \t
  • repeat this for different scales with some nice blending!
__code_box( GLSL, vec4
truchet(vec2 p, float r, float w)
{
\tfloat hr = r / 2.;
\tfloat hw = w / 2.;

\tvec2 b = floor(p / r) * r;
\tvec2 c = b + vec2(hr);
\tuint h = hash(uvec2(abs(c * 100.))) % 4u;

\tvec2 a[] = vec2[](
\t\tb + vec2(r, 0),
\t\tb + vec2(r),
\t\tb + vec2(0, r),
\t\tb,
\t\tb + vec2(0, r),
\t\tb,
\t\tb + vec2(r, 0),
\t\tb + vec2(r)
\t);

\tfloat d = distance(p, a[h]);
\tif (abs(d - hr) < hw) {
\t\treturn vec4(1.);
\t}

\tif (abs(d - hr) < w * 1.5) {
\t\treturn vec4(color(itime * 0.5), (1. - (abs(d - hr) - hw) / w) * 0.5);
\t}

\td = distance(p, a[h + 4u]);
\tif (abs(d - hr) < hw) {
\t\treturn vec4(1.);
\t}

\t\treturn vec4(color(itime * 0.5), (1. - (abs(d - hr) - hw) / w) * 0.5);

truchet(vec2 p, float r, float w)
{
\tfloat hr = r / 2.;
\tfloat hw = w / 2.;

\tvec2 b = floor(p / r) * r;
\tvec2 c = b + vec2(hr);
\tfloat h = float(hash(uvec2(abs(c * 100.))) % 4u);
\tfloat i = 3. - h;

\tvec2 a0 = b + vec2(mod(h, 2.) * r, floor(h / 2.) * r);
\tvec2 a1 = b + vec2(mod(i, 2.) * r, floor(i / 2.) * r);

\tfloat d1 = distance(p, a0);
\tfloat d2 = distance(p, a1);

\tfloat d = min(d1, d2);

\tvec4 glow = vec4(
\t\tcolor(triangle(itime * 0.125, 3., 1.)),
\t\t(1. - (abs(d - hr) - hw) / w) * 0.5
\t);

\treturn mix(vec4(1.), glow, smoothstep(hw, hw + 1.414, abs(d - hr)));
} ) __p(alright, so i got rid of the array! instead, we have some math. the idea is that we pick a hash the same way as above, but instead of indexing into an array, we treat that hash as one of the four corners of our cell! then, once we pick one corner of our cell, we can pick the diagonally opposing corner using __cs(3 - h) where __cs(h) is the first corner we picked.) __oimg(./assets/truchet_expl.webp, explanation of the calculation for the opposite corner. we simply subtract the corner index from 3.) __h3(final product) __p(check out the full shader on __a(shadertoy, https:\/\/www.shadertoy.com\/view\/tctGRf)!) __ending
